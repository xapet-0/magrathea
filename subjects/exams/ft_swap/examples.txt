Examples:
    int a = 1, b = 2; ft_swap(&a, &b); printf("%d %d\n", a, b)
    Output: 2 1
    int a = -10, b = -20; ft_swap(&a, &b); printf("%d,%d\n", a, b)
    Output: -20,-10
    int a = 42, b = 43; ft_swap(&a, &b); printf("%d,%d\n", a, b)
    Output: 43,42
    int a = 0, b = 1337; ft_swap(&a, &b); printf("%d,%d\n", a, b)
    Output: 1337,0
