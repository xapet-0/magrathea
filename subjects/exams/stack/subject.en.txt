Assignment name  : stack
Expected files   : stack.c
Allowed functions: malloc, free
--------------------------------------------------------------------------------

Implement a LIFO stack data structure with the following 5 functions:
struct s_stack *init(void);
void *pop(struct s_stack *stack);
void push(struct s_stack *stack, void *content);
void *peek(struct s_stack *stack);
int isEmpty(struct s_stack *stack);

Required structure / declarations:
struct s_node { void *content; struct s_node *next; }; struct s_stack { struct s_node *top; }; struct s_stack *init(void); void *pop(struct s_stack *stack); void push(struct s_stack *stack, void *content); void *peek(struct s_stack *stack); int isEmpty(struct s_stack *stack);

Examples:
    struct s_stack *s = init(); push(s, "42"); char *v1 = (char *)pop(s); int e1 = isEmpty(s); printf("%s,%d\n", v1, e1)
    Output: 42,1
    struct s_stack *s = init(); printf("%d,%d,%d\n", isEmpty(s), peek(s) == NULL, pop(s) == NULL)
    Output: 1,1,1
    struct s_stack *s = init(); push(s, "a"); push(s, "b"); char *p1 = (char *)peek(s); char *d1 = (char *)pop(s); char *d2 = (char *)pop(s); int e2 = isEmpty(s); printf("%s,%s,%s,%d\n", p1, d1, d2, e2)
    Output: b,b,a,1
