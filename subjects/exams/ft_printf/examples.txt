Examples:
    int r = ft_printf("Hex %s is %d and %x\n", "number", 42, 42); printf("%d\n", r);
    Output: Hex number is 42 and 2a | 24
    int r = ft_printf("%s\n", (char *)0); printf("%d\n", r);
    Output: (null) | 7
    int r = ft_printf("%d %d %x\n", 0, -2147483648, 0); printf("%d\n", r);
    Output: 0 -2147483648 0 | 16
    int r = ft_printf("1337"); printf(":%d\n", r);
