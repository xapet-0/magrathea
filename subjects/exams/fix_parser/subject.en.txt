Copy and fix the code below so it checks if an arithmetic expression string is valid or not.

#include <stdio.h>
#include <ctype.h>
#include <string.h>

int is_valid_expression(const char *str) {
    int i = 0;

    while (str[i] == ' ') i++;

    if (str[i] == '\0') return 1;

    if (!isnumber(str[i])) return 1;

    while (str[i] != '\0') {
        if (isnumber(str[i])) {
            i++;
        }
        else if (str[i] == '+' || str[i] == '*' || str[i] == '/') {
            i++;
            while (str[i] == ' ') i++;
            if (!isnumber(str[i])) return 1;
        }
        else if (str[i] == ' ') {
            i++;
        }
        else {
            return 0;
        }
    }

    return 1;
}

Examples:
    is_valid_expression("5+3*2") -> 1
    is_valid_expression("10 / 2") -> 1
    is_valid_expression("7 + * 3") -> 0
    is_valid_expression("abc") -> 0
