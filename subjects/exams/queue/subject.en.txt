Assignment name  : queue
Expected files   : queue.c
Allowed functions: malloc, free
--------------------------------------------------------------------------------

Implement a FIFO queue data structure with the following 5 functions:
struct s_queue *init(void);
void enqueue(struct s_queue *queue, void *content);
void *dequeue(struct s_queue *queue);
void *peek(struct s_queue *queue);
int isEmpty(struct s_queue *queue);

Required structure / declarations:
struct s_node { void *content; struct s_node *next; }; struct s_queue { struct s_node *first; struct s_node *last; }; struct s_queue *init(void); void enqueue(struct s_queue *queue, void *content); void *dequeue(struct s_queue *queue); void *peek(struct s_queue *queue); int isEmpty(struct s_queue *queue);

Examples:
    struct s_queue *q = init(); enqueue(q, "1337"); char *v1 = (char *)dequeue(q); int e1 = isEmpty(q); printf("%s,%d\n", v1, e1)
    Output: 1337,1
    struct s_queue *q = init(); printf("%d,%d,%d\n", isEmpty(q), peek(q) == NULL, dequeue(q) == NULL)
    Output: 1,1,1
    struct s_queue *q = init(); enqueue(q, "a"); enqueue(q, "b"); char *p1 = (char *)peek(q); char *d1 = (char *)dequeue(q); char *d2 = (char *)dequeue(q); int e2 = isEmpty(q); printf("%s,%s,%s,%d\n", p1, d1, d2, e2)
    Output: a,a,b,1
